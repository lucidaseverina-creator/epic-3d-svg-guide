type V3 = { x: number; y: number; z: number };

function lerp(a: V3, b: V3, t: number): V3 {
  return { x: a.x + (b.x - a.x) * t, y: a.y + (b.y - a.y) * t, z: a.z + (b.z - a.z) * t };
}

interface SharedBoundary {
  // Canonical eval: u in [0,1). Internally this is PANP piecewise mapping across port segments.
  eval(u: number): V3;

  // Stable seam + port->u pins live here.
  // e.g. { commissure_L: 0.0, upper_mid: 0.25, commissure_R: 0.5, lower_mid: 0.75 }
  portU: Record<string, number>;
}

function buildAnnularQuadPatch(params: {
  outer: SharedBoundary;
  inner: SharedBoundary;
  M: number;   // around loop
  H: number;   // radial
}) {
  const { outer, inner, M, H } = params;

  // --- Sampler frames (no-crack contract)
  const O: V3[] = new Array(M);
  const I: V3[] = new Array(M);
  for (let k = 0; k < M; k++) {
    const u = k / M;
    O[k] = outer.eval(u);
    I[k] = inner.eval(u);
  }

  // --- Vertex grid
  const verts: number[] = []; // xyz packed
  for (let r = 0; r <= H; r++) {
    const v = r / H;
    for (let k = 0; k < M; k++) {
      const p = lerp(O[k], I[k], v);
      verts.push(p.x, p.y, p.z);
    }
  }

  // --- Quad indices as triangles (2 per quad)
  const idx: number[] = [];
  const row = M;
  for (let r = 0; r < H; r++) {
    for (let k = 0; k < M; k++) {
      const k1 = (k + 1) % M;

      const a = r * row + k;
      const b = r * row + k1;
      const c = (r + 1) * row + k1;
      const d = (r + 1) * row + k;

      // (a,b,c) and (a,c,d)
      idx.push(a, b, c, a, c, d);
    }
  }

  return { positions: new Float32Array(verts), indices: new Uint32Array(idx) };
}

export function pxToMeters(px: Px, cal: ViewCal) {
  const mpp = 1 / cal.pxPerMeter;
  const u = (px[0] - cal.originPx[0]) * mpp;
  const v = (px[1] - cal.originPx[1]) * mpp;
  return { u, v };
}

export function centerlineAngle(cal: ViewCal) {
  const [a, b] = cal.centerlinePx;
  const dx = b[0] - a[0];
  const dy = b[1] - a[1];
  return Math.atan2(dy, dx); // image-space angle
}

export function rotate2(u: number, v: number, ang: number) {
  const c = Math.cos(ang), s = Math.sin(ang);
  return { u: u * c - v * s, v: u * s + v * c };
}

// refpack/ReferencePlane.tsx
import * as THREE from "three";
import { useMemo } from "react";
import { useLoader } from "@react-three/fiber";
import type { ViewDef, ViewAxis } from "./types";
import { centerlineAngle } from "./math";

function desiredAngle(axis: ViewAxis) {
  // TOP/SIDE: want centerline horizontal (+u). FRONT/REAR: want centerline vertical (+v) in image space.
  return (axis === "TOP" || axis === "SIDE") ? 0 : Math.PI / 2;
}

function baseOrientation(axis: ViewAxis) {
  // Start from PlaneGeometry in XY with normal +Z.
  // Return Euler to orient plane into world “view plane”.
  switch (axis) {
    case "TOP":   return new THREE.Euler(-Math.PI / 2, 0, 0);      // XY -> XZ (normal +Y)
    case "SIDE":  return new THREE.Euler(0, 0, 0);                 // XY at constant Z
    case "FRONT": return new THREE.Euler(0,  Math.PI / 2, 0);      // XY -> YZ (normal +X)
    case "REAR":  return new THREE.Euler(0, -Math.PI / 2, 0);      // XY -> YZ (normal -X)
  }
}

export function ReferencePlane(props: {
  view: ViewDef;
  imageUrl: string;              // resolved absolute URL for texture
  imageSizePx: { w: number; h: number };
  constantPos: number;           // Y for TOP, Z for SIDE, X for FRONT/REAR
}) {
  const { view, imageUrl, imageSizePx, constantPos } = props;
  const tex = useLoader(THREE.TextureLoader, imageUrl);

  const geom = useMemo(() => {
    const mpp = 1 / view.cal.pxPerMeter;
    const Wm = imageSizePx.w * mpp;
    const Hm = imageSizePx.h * mpp;
    return new THREE.PlaneGeometry(Wm, Hm);
  }, [imageSizePx.w, imageSizePx.h, view.cal.pxPerMeter]);

  const mat = useMemo(() => {
    tex.flipY = false;
    tex.needsUpdate = true;
    return new THREE.MeshBasicMaterial({
      map: tex,
      transparent: true,
      opacity: view.opacity,
      side: THREE.DoubleSide,
      depthWrite: false,
    });
  }, [tex, view.opacity]);

  const rot = useMemo(() => {
    const theta = centerlineAngle(view.cal);
    const r = desiredAngle(view.axis) - theta; // rotate so centerline aligns to desired axis
    const base = baseOrientation(view.axis);
    // Apply as Z-rotation in local plane space AFTER base orientation:
    return new THREE.Euler(base.x, base.y, base.z + r);
  }, [view.axis, view.cal]);

  const pos = useMemo<[number, number, number]>(() => {
    switch (view.axis) {
      case "TOP":   return [0, constantPos, 0];
      case "SIDE":  return [0, 0, constantPos];
      case "FRONT": return [constantPos, 0, 0];
      case "REAR":  return [constantPos, 0, 0];
    }
  }, [view.axis, constantPos]);

  return <mesh geometry={geom} material={mat} position={pos} rotation={rot} />;
}

// panp_sampler.ts
import { Vec3, V3, Curve3D, Port } from "./topo_types";

export type BoundaryFrame = {
  k: number;      // sample index
  u: number;      // [0,1) around loop (canonical)
  p: Vec3;        // position
  // Optional: additional per-frame data (tangent, normal, etc.)
};

export type SharedBoundarySampler = {
  M: number;
  frames: BoundaryFrame[];
  // helper: find exact frame indices near a u-span (optional)
  kRangeForU?: (uLo: number, uHi: number) => { k0: number; k1: number };
};

// Build a u-grid that is deterministic and respects pinned ports.
// - includes seam u=0
// - includes all pinned u values
// - fills between pins with uniform subdivisions
export function buildPinnedUGrid(params: {
  M: number;                 // desired around-loop resolution
  ports: Port[];             // ports with uPin
  seamUPin: number;          // usually 0
}): number[] {
  const { M, ports, seamUPin } = params;

  const pins = new Set<number>();
  pins.add(wrap01(seamUPin));

  for (const p of ports) if (p.uPin !== undefined) pins.add(wrap01(p.uPin));

  // Sort pins in [0,1)
  const pinList = [...pins].sort((a, b) => a - b);

  // Ensure 0 is first (canonical seam)
  if (pinList[0] !== 0) {
    // rotate list so that 0 is first if present; else insert 0
    if (pinList.includes(0)) {
      while (pinList[0] !== 0) pinList.push(pinList.shift()!);
    } else {
      pinList.unshift(0);
    }
  }

  // Allocate samples across spans proportional to span length
  const spans: { a: number; b: number; len: number }[] = [];
  for (let i = 0; i < pinList.length; i++) {
    const a = pinList[i];
    const b = pinList[(i + 1) % pinList.length];
    const len = (b >= a) ? (b - a) : (1 - a + b);
    spans.push({ a, b, len });
  }

  const totalLen = spans.reduce((s, t) => s + t.len, 0) || 1;
  const rawCounts = spans.map(sp => sp.len / totalLen * M);

  // Deterministic rounding: floor + distribute remainder by largest fractional parts
  const counts = rawCounts.map(Math.floor);
  let used = counts.reduce((s, c) => s + c, 0);
  let rem = Math.max(0, M - used);

  const frac = rawCounts.map((v, i) => ({ i, f: v - Math.floor(v) }))
                        .sort((a, b) => b.f - a.f);
  for (let r = 0; r < rem; r++) counts[frac[r % frac.length].i]++;

  // Build grid: include pin at start of each span, then interior points (excluding end pin)
  const grid: number[] = [];
  for (let i = 0; i < spans.length; i++) {
    const { a, b } = spans[i];
    const n = Math.max(1, counts[i]); // at least 1 step per span
    for (let j = 0; j < n; j++) {
      const t = j / n; // excludes end pin (t=1)
      const u = wrap01(a + spanDelta(a, b) * t);
      grid.push(u);
    }
  }

  // Ensure exact seam 0 exists as first element
  grid.sort((x, y) => x - y);
  // rotate so first is 0
  while (grid.length && grid[0] !== 0) grid.push(grid.shift()!);

  return grid;
}

function spanDelta(a: number, b: number): number {
  return (b >= a) ? (b - a) : (1 - a + b);
}
function wrap01(u: number): number {
  u = u % 1;
  if (u < 0) u += 1;
  // snap tiny floating noise
  if (Math.abs(u) < 1e-12) return 0;
  if (Math.abs(u - 1) < 1e-12) return 0;
  return u;
}

// Sample a closed boundary by u-grid, using a canonical curve C3D(t) plus an optional u->t map.
// If you already define u == t (common for authored loops), just use t=u.
export function sampleBoundaryByPinnedU(params: {
  M: number;
  uGrid: number[];
  curve: Curve3D;
  uToT?: (u: number) => number;
}): SharedBoundarySampler {
  const { M, uGrid, curve } = params;
  const uToT = params.uToT ?? ((u: number) => u);

  const frames: BoundaryFrame[] = [];
  for (let k = 0; k < uGrid.length; k++) {
    const u = uGrid[k];
    const t = uToT(u);
    frames.push({ k, u, p: curve.eval(t) });
  }

  return { M, frames };
}

import { MeshData, Quad } from "./mesh_types";
import { Vec2, Vec3, V3 } from "./topo_types";
import { SharedBoundary, rebuildBoundarySampler } from "./shared_boundary";

export type PatchInteriorSolver = (params: {
  outerPoint: Vec3;
  innerPoint: Vec3;
  u: number;     // around
  v: number;     // radial
}) => Vec3;

// Default interior: linear in 3D (NOT on-surface for curved underforms; use only as placeholder)
export const linearInterior: PatchInteriorSolver = ({ outerPoint, innerPoint, v }) =>
  V3.lerp(outerPoint, innerPoint, v);

export function emitAnnularQuadPatch(params: {
  outer: SharedBoundary;
  inner: SharedBoundary;
  M: number;     // around resolution
  N: number;     // radial resolution (>=2)
  solver?: PatchInteriorSolver;
}): MeshData {
  const { outer, inner, M, N } = params;
  const solver = params.solver ?? linearInterior;

  const so = outer.sampler ?? rebuildBoundarySampler(outer, M);
  const si = inner.sampler ?? rebuildBoundarySampler(inner, M);

  if (so.frames.length !== si.frames.length) {
    throw new Error(`AnnularQuadPatch requires matched u-grid. Outer=${so.frames.length}, Inner=${si.frames.length}`);
  }

  const positions: Vec3[] = [];
  const uvs: Vec2[] = [];
  const quads: Quad[] = [];

  // Vertex indexing: idx(uIndex, vIndex) where uIndex in [0..M-1], vIndex in [0..N-1]
  const idx = (uI: number, vI: number) => uI * N + vI;

  for (let uI = 0; uI < so.frames.length; uI++) {
    const u = so.frames[uI].u;
    const pOuter = so.frames[uI].p;
    const pInner = si.frames[uI].p;

    for (let vI = 0; vI < N; vI++) {
      const v = (N === 1) ? 0 : vI / (N - 1);
      const p = solver({ outerPoint: pOuter, innerPoint: pInner, u, v });

      positions.push(p);
      uvs.push({ x: u, y: v });
    }
  }

  // Quads: (uI,vI) cell => four corners
  for (let uI = 0; uI < so.frames.length; uI++) {
    const uN = (uI + 1) % so.frames.length;
    for (let vI = 0; vI < N - 1; vI++) {
      const a = idx(uI, vI);
      const b = idx(uN, vI);
      const c = idx(uN, vI + 1);
      const d = idx(uI, vI + 1);
      quads.push([a, b, c, d]);
    }
  }

  return { positions, uvs, quads };
}

4) Sphere milestone: exact annular patch (no approximation)

Your “prove it on a sphere” milestone is perfect because we can make the interior solver exact.

If the user draws circles in the TOP ortho (radius = a), those are cylinders intersecting the sphere, producing latitude circles. We can represent the boundary exactly and fill the annulus exactly with spherical coordinates.

// sphere_annulus_exact.ts
import { Vec3 } from "./topo_types";
import { PatchInteriorSolver } from "./annular_quad_patch";

// Exact sphere surface point for (u around, theta polar)
export function spherePoint(r: number, u: number, theta: number): Vec3 {
  const phi = 2 * Math.PI * u;
  const s = Math.sin(theta);
  const c = Math.cos(theta);
  return {
    x: r * s * Math.cos(phi),
    y: r * c,
    z: r * s * Math.sin(phi),
  };
}

// Convert TOP-drawn circle radius a to polar theta where sin(theta)=a/r
export function thetaFromTopRadius(r: number, a: number): number {
  const t = Math.asin(clamp(a / r, -1, 1));
  // theta in [0,pi], asin gives [-pi/2,pi/2]; pick top hemisphere band if desired
  return t; // top hemisphere near +y if theta small? (depends on your convention)
}

function clamp(x: number, a: number, b: number): number {
  return Math.max(a, Math.min(b, x));
}

// Exact solver for AnnularQuadPatch on a sphere annulus between thetaOuter and thetaInner.
// (u is shared; v lerps theta, guaranteeing points stay on sphere)
export function makeSphereAnnulusSolver(r: number, thetaOuter: number, thetaInner: number): PatchInteriorSolver {
  return ({ u, v }) => {
    const theta = thetaOuter + (thetaInner - thetaOuter) * v;
    return spherePoint(r, u, theta);
  };
}

// topo_update_pipeline.ts
import { SharedBoundary, rebuildBoundarySampler } from "./shared_boundary";
import { emitAnnularQuadPatch } from "./annular_quad_patch";

export type PatchId = string;

export type PatchNode =
  | { id: PatchId; type: "AnnularQuadPatch"; outer: SharedBoundary; inner: SharedBoundary; M: number; N: number; };

export type PatchGraph = {
  patches: PatchNode[];
  // dependency index: boundaryId -> patchIds
  deps: Map<string, PatchId[]>;
};

export function rebuildAfterBoundaryEdit(params: {
  boundaryIdsDirty: string[];
  graph: PatchGraph;
  samplerM: number;
}): Map<PatchId, any /*MeshData*/> {
  const { boundaryIdsDirty, graph, samplerM } = params;

  // 1) rebuild samplers for dirty SharedBoundaries
  for (const bid of boundaryIdsDirty) {
    // find boundary objects referenced by patches (simple approach)
    for (const p of graph.patches) {
      if (p.type === "AnnularQuadPatch") {
        if (p.outer.id === bid) rebuildBoundarySampler(p.outer, samplerM);
        if (p.inner.id === bid) rebuildBoundarySampler(p.inner, samplerM);
      }
    }
  }

  // 2) rebuild only dependent patches
  const out = new Map<PatchId, any>();
  const patchIds = new Set<PatchId>();
  for (const bid of boundaryIdsDirty) {
    const ds = graph.deps.get(bid) ?? [];
    for (const pid of ds) patchIds.add(pid);
  }

  for (const p of graph.patches) {
    if (!patchIds.has(p.id)) continue;

    if (p.type === "AnnularQuadPatch") {
      const mesh = emitAnnularQuadPatch({
        outer: p.outer,
        inner: p.inner,
        M: p.M,
        N: p.N,
        // solver plugged elsewhere (sphere exact / general relaxation)
      });
      out.set(p.id, mesh);
    }
  }

  return out;
}

  id: string;
  ports: Port[];                          // hard u locks
  eval(u: number): Vec3;                  // exact on-surface point
  // optional: tangent(u), normal(u), etc.
};

type AnnularQuadPatch = {
  id: string;
  outer: SharedBoundary;
  inner: SharedBoundary;
  M: number; // columns around (u samples)
  N: number; // radial rows (v samples)
  // eval(u,v) may be direct (analytic) or solved via constraint lift
  eval(u: number, v: number): Vec3;
};

function emitAnnularQuads(p: AnnularQuadPatch) {
  const { M, N } = p;
  const verts: Vec3[] = [];
  const idx = (r:number,k:number) => r*M + ((k%M)+M)%M;

  for (let r=0; r<=N; r++) {
    const v = r / N;
    for (let k=0; k<M; k++) {
      const u = k / M;         // u=0 is seam by construction (port-locked)
      verts.push(p.eval(u, v));
    }
  }

  const quads: [number,number,number,number][] = [];
  for (let r=0; r<N; r++) {
    for (let k=0; k<M; k++) {
      const a = idx(r,   k);
      const b = idx(r,   k+1);
      const c = idx(r+1, k+1);
      const d = idx(r+1, k);
      quads.push([a,b,c,d]);   // deterministic quad grid
    }
  }

  return { verts, quads };
}

import * as THREE from "three";
import { useMemo } from "react";
import { useLoader } from "@react-three/fiber";

type Px = [number, number];
type ViewAxis = "TOP" | "SIDE" | "FRONT" | "REAR";

interface ViewCal {
  pxPerMeter: number;
  originPx: Px;
  centerlinePx: [Px, Px];
  anchorA_Px: Px;
  anchorB_Px: Px;
}

interface ViewDef {
  imageUrl: string;
  axis: ViewAxis;
  cal: ViewCal;
  opacity: number;
  crop?: { x: number; y: number; width: number; height: number } | null;
}

function centerlineAngle(cal: ViewCal): number {
  const [a, b] = cal.centerlinePx;
  return Math.atan2(b[1] - a[1], b[0] - a[0]);
}

export function ReferencePlane({
  view,
  imageSizePx,     // { w: number; h: number } from image metadata
  constantPos,     // e.g. Y_max+margin for TOP, etc.
}: {
  view: ViewDef;
  imageSizePx: { w: number; h: number };
  constantPos: number;
}) {
  const texture = useLoader(THREE.TextureLoader, view.imageUrl);

  const { planeWm, planeHm, rotZ, worldQuat, worldPos } = useMemo(() => {
    const mpp = 1 / view.cal.pxPerMeter;
    const planeWm = imageSizePx.w * mpp;
    const planeHm = imageSizePx.h * mpp;

    const theta = centerlineAngle(view.cal);
    const rotZ = -theta; // rotate image so centerline becomes horizontal

    // Orient plane into world depending on axis (so its local u/v match expected mapping)
    // We'll build a quaternion that maps a default XY plane into the correct world plane.
    const q = new THREE.Quaternion();
    const pos = new THREE.Vector3();

    switch (view.axis) {
      case "TOP":
        // default PlaneGeometry is in XY; we want u->X and v->Z, plane at Y=constantPos
        q.setFromEuler(new THREE.Euler(-Math.PI / 2, 0, rotZ));
        pos.set(0, constantPos, 0);
        break;
      case "SIDE":
        // want u->X and v->Y, plane at Z=constantPos
        q.setFromEuler(new THREE.Euler(0, 0, rotZ));
        pos.set(0, 0, constantPos);
        break;
      case "FRONT":
        // want u->Z and v->Y, plane at X=constantPos
        q.setFromEuler(new THREE.Euler(0, Math.PI / 2, 0)); // rotate XY plane so X->Z
        // rotZ could be applied around plane normal if needed later
        pos.set(constantPos, 0, 0);
        break;
      case "REAR":
        q.setFromEuler(new THREE.Euler(0, -Math.PI / 2, 0));
        pos.set(constantPos, 0, 0);
        break;
    }

    return { planeWm, planeHm, rotZ, worldQuat: q, worldPos: pos };
  }, [view, imageSizePx, constantPos]);

  texture.flipY = false;

  const geom = useMemo(() => new THREE.PlaneGeometry(planeWm, planeHm), [planeWm, planeHm]);

  const mat = useMemo(
    () =>
      new THREE.MeshBasicMaterial({
        map: texture,
        transparent: true,
        opacity: view.opacity,
        side: THREE.DoubleSide,
        depthWrite: false,
      }),
    [texture, view.opacity]
  );

  return (
    <mesh geometry={geom} material={mat} position={worldPos} quaternion={worldQuat} />
  );
}

// Convert plane-local (u right, v down) in meters -> pixel coords
export function uvMetersToPx(u: number, v: number, cal: ViewCal): Px {
  // u,v are meters in the *rotated* metric plane basis (after centerline alignment),
  // but since we also rotate the plane mesh by that same angle, local coords already match.
  const x = cal.originPx[0] + u * cal.pxPerMeter;
  const y = cal.originPx[1] + v * cal.pxPerMeter;
  return [x, y];
}

// Convert pixel coords -> plane-local meters (u right, v down)
export function pxToUvMeters(px: Px, cal: ViewCal): { u: number; v: number } {
  const mpp = metersPerPx(cal);
  return {
    u: (px[0] - cal.originPx[0]) * mpp,
    v: (px[1] - cal.originPx[1]) * mpp,
  };
}

// Plane-local coordinates from raycast point
// Assumes plane geometry is XY, local +x right, local +y up.
// We choose: u = x, v = -y (so v is down, matching image convention).
export function localPointToUvMeters(localPt: THREE.Vector3): { u: number; v: number } {
  return { u: localPt.x, v: -localPt.y };
}

export function computePxPerMeter(a: Px, b: Px, meters: number): number {
  const dx = b[0] - a[0];
  const dy = b[1] - a[1];
  const dpx = Math.sqrt(dx * dx + dy * dy);

Curtain intersection solver (stable, branch-tracked)

We solve along the view-normal direction so that the curve stays exactly what it was in the drawing plane.

// src/model/cutCurtainIntersect.ts
import * as THREE from "three";
import { Field } from "./field";
import { Curve2D, SurfaceCurve } from "./curves";

type Curtain = {
  origin: (t: number) => THREE.Vector3; // base point on curtain for param t
  dir: THREE.Vector3;                   // curtain extrusion direction (unit)
};

function newtonOnRay(params: {
  field: Field;
  ro: THREE.Vector3;
  rd: THREE.Vector3;
  t0: number;               // initial distance along ray
  maxIter?: number;
  tol?: number;
}): number {
  const { field, ro, rd } = params;
  let x = params.t0;
  const maxIter = params.maxIter ?? 30;
  const tol = params.tol ?? 1e-10;

  for (let i = 0; i < maxIter; i++) {
    const p = ro.clone().addScaledVector(rd, x);
    const f = field.eval(p);
    if (Math.abs(f) < tol) return x;

    const g = field.grad(p);
    const df = g.dot(rd); // derivative along ray
    if (Math.abs(df) < 1e-12) break; // fallback to safer method upstream

    x = x - f / df;
  }
  return x;
}

// A safer bracket+Newton hybrid for ugly cases.
function solveRaySDF(params: {
  field: Field;
  ro: THREE.Vector3;
  rd: THREE.Vector3;
  tPrev?: number;
}): number {
  const { field, ro, rd, tPrev } = params;

  // 1) Use continuity: previous solution is the best branch selector.
  let t0 = tPrev ?? 0;

  // 2) Quick Newton.
  let t = newtonOnRay({ field, ro, rd, t0 });
  const p = ro.clone().addScaledVector(rd, t);
  if (Math.abs(field.eval(p)) < 1e-8) return t;

  // 3) If Newton didn’t settle, do a small bracket search around t0.
  let a = t0 - 2;
  let b = t0 + 2;
  let fa = field.eval(ro.clone().addScaledVector(rd, a));
  let fb = field.eval(ro.clone().addScaledVector(rd, b));

  // expand bracket deterministically
  for (let k = 0; k < 20 && fa * fb > 0; k++) {
    a -= 2;
    b += 2;
    fa = field.eval(ro.clone().addScaledVector(rd, a));
    fb = field.eval(ro.clone().addScaledVector(rd, b));
  }
  if (fa * fb > 0) {
    throw new Error("Curtain intersection: could not bracket a root (curve misses surface).");
  }

  // bisection
  for (let it = 0; it < 60; it++) {
    const m = 0.5 * (a + b);
    const fm = field.eval(ro.clone().addScaledVector(rd, m));
    if (Math.abs(fm) < 1e-10) return m;
    if (fa * fm <= 0) {
      b = m; fb = fm;
    } else {
      a = m; fa = fm;
    }
  }
  return 0.5 * (a + b);
}

export function makeSurfaceCurveFromCurtain(params: {
  id: string;
  curve2D: Curve2D;
  curtain: Curtain;
  field: Field;
}): SurfaceCurve {
  let tPrev: number | undefined = undefined;

  return {
    id: params.id,
    ports: params.curve2D.ports,
    eval: (t: number) => {
      const base = params.curtain.origin(t);        // exactly preserves 2D curve
      const dist = solveRaySDF({ field: params.field, ro: base, rd: params.curtain.dir, tPrev });
      tPrev = dist;
      return base.clone().addScaledVector(params.curtain.dir, dist);
    },
  };
}


This is the key “no drift” mechanism: the curve is always computed from (SVG curve) ∩ (field) with branch continuity, never baked.

// src/model/patch/templates/annularQuad.ts
import * as THREE from "three";
import { Field } from "../../field";
import { SurfaceCurve } from "../../curves";

export type QuadMesh = {
  positions: Float32Array;
  quads: Uint32Array; // 4 indices per quad
};

// project an initial guess back onto the implicit surface
function snapToSurface(field: Field, p0: THREE.Vector3): THREE.Vector3 {
  let p = p0.clone();
  for (let i = 0; i < 20; i++) {
    const f = field.eval(p);
    if (Math.abs(f) < 1e-10) return p;
    const g = field.grad(p);
    const gg = g.lengthSq();
    if (gg < 1e-18) break;
    // Newton step along gradient
    p = p.addScaledVector(g, -f / gg);
  }
  return p;
}

export function buildAnnularQuadPatch(params: {
  field: Field;
  outer: SurfaceCurve;     // B0
  inner: SurfaceCurve;     // B1
  M: number;
  R: number;
}): QuadMesh {
  const { field, outer, inner, M, R } = params;

  const V = (R + 1) * M;
  const positions = new Float32Array(V * 3);

  // helper: wrap i
  const idx = (r: number, i: number) => r * M + ((i % M) + M) % M;

  // 1) sample both loops consistently by parameter t in [0,1)
  // In your full system: use PANP + ports to set seam t=0 and arc-length for uniform spacing.
  // For this milestone: uniform t is fine; the *topological* behavior is the point.
  for (let i = 0; i < M; i++) {
    const t = i / M;
    const po = outer.eval(t);
    const pi = inner.eval(t);

    for (let r = 0; r <= R; r++) {
      const u = r / R; // 0..1 radial blend

      // initial guess in 3D
      const pGuess = po.clone().lerp(pi, u);

      // snap exactly to the surface field
      const p = snapToSurface(field, pGuess);

      const v = idx(r, i);
      positions[3 * v + 0] = p.x;
      positions[3 * v + 1] = p.y;
      positions[3 * v + 2] = p.z;
    }
  }

  // 2) emit quads
  const Q = R * M;
  const quads = new Uint32Array(Q * 4);
  let q = 0;

  for (let r = 0; r < R; r++) {
    for (let i = 0; i < M; i++) {
      const a = idx(r, i);
      const b = idx(r, i + 1);
      const c = idx(r + 1, i + 1);
      const d = idx(r + 1, i);

      quads[4 * q + 0] = a;
      quads[4 * q + 1] = b;
      quads[4 * q + 2] = c;
      quads[4 * q + 3] = d;
      q++;
    }
  }

  return { positions, quads };
}

// src/model/fields/sphereField.ts
import * as THREE from "three";
import { Field } from "../field";

export function makeSphereField(center: THREE.Vector3, r: number): Field {
  return {
    eval: (p) => p.clone().sub(center).length() - r,
    grad: (p) => {
      const v = p.clone().sub(center);
      const len = v.length();
      if (len < 1e-12) return new THREE.Vector3(0, 1, 0);
      return v.multiplyScalar(1 / len);
    },
  };
